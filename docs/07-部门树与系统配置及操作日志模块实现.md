## 9 部门树、系统配置与操作日志模块

本章讲三个辅助模块：部门表示组织归属，系统配置提供可在线调整的运行参数，操作日志提供事后追溯依据。三者都不直接面向业务单据，但决定了系统“能不能按组织分工运转”“参数改了要不要重启”“出问题能不能查到人”。实现上共同遵守两条原则：写操作走事务，关键动作必留痕。

### 9.1 部门树管理

**数据结构：**部门表用 parent_id 表示层级，根节点的 parent_id 为空而不是 0，这样“未设置上级”与“上级编号为 0”不会混淆；排序字段 sort_no 决定同级展示顺序。树不在数据库里递归查询，而是把全部记录一次取出后由业务层组装，部门数量在几十条量级，一次查询加内存组装的成本远低于递归多条 SQL。

**增删改的三条约束：**新增与修改都要过三道校验——部门编码全局唯一、同级不能重名、上级必须真实存在；修改时额外做“防环”检查：从目标上级逐级向上追溯，一旦发现当前部门出现在祖先链中即拒绝，否则把父级改成自己的下级会让整棵树脱离根节点、页面上永远展不开；删除前必须确认没有子部门、没有在岗成员，并且校验与删除放在同一事务内，防止校验通过后被并发新增的子部门漏判。所有校验失败都返回可直接展示给用户的中文提示。

```go
package service

import (
	"errors"

	"gorm.io/gorm"

	"industrial_admin/model"
	"industrial_admin/response"
	"industrial_admin/utils"
)

type DeptService struct{ db *gorm.DB }

func NewDeptService(database *gorm.DB) *DeptService { return &DeptService{db: database} }

// Tree 取全部部门并组装为树，前端直接渲染
func (s *DeptService) Tree() ([]*utils.TreeNode, error) {
	var depts []model.Dept
	if err := s.db.Order("sort_no ASC, id ASC").Find(&depts).Error; err != nil {
		return nil, utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}

	nodes := make([]*utils.TreeNode, 0, len(depts))
	for _, d := range depts {
		var parent uint64
		if d.ParentID != nil {
			parent = *d.ParentID
		}
		nodes = append(nodes, &utils.TreeNode{
			ID: d.ID, ParentID: parent, Name: d.DeptName,
			Code: d.DeptCode, SortNo: d.SortNo,
		})
	}
	return utils.BuildTree(nodes), nil
}

type CreateDeptReq struct {
	ParentID *uint64 `json:"parentId"`
	DeptName string  `json:"deptName" binding:"required"`
	DeptCode string  `json:"deptCode" binding:"required"`
	LeaderID *uint64 `json:"leaderId"`
	SortNo   int     `json:"sortNo"`
	Remark   string  `json:"remark"`
}

// Create 新增部门：编码唯一，同级名称唯一，上级必须存在
func (s *DeptService) Create(req *CreateDeptReq) (uint64, error) {
	if req.ParentID != nil {
		var parentCount int64
		if err := s.db.Model(&model.Dept{}).Where("id = ?", *req.ParentID).Count(&parentCount).Error; err != nil {
			return 0, utils.WrapAppError(response.CodeServerError, "查询上级部门失败", err)
		}
		if parentCount == 0 {
			return 0, utils.NewAppError(response.CodeParamInvalid, "上级部门不存在")
		}
	}

	var dup int64
	if err := s.db.Model(&model.Dept{}).Where("dept_code = ?", req.DeptCode).Count(&dup).Error; err != nil {
		return 0, utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}
	if dup > 0 {
		return 0, utils.NewAppError(response.CodeConflict, "部门编码已存在")
	}

	// 同级名称唯一：由业务层校验，数据库不建唯一约束
	scope := s.db.Model(&model.Dept{}).Where("dept_name = ?", req.DeptName)
	if req.ParentID != nil {
		scope = scope.Where("parent_id = ?", *req.ParentID)
	} else {
		scope = scope.Where("parent_id IS NULL")
	}
	var sameName int64
	if err := scope.Count(&sameName).Error; err != nil {
		return 0, utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}
	if sameName > 0 {
		return 0, utils.NewAppError(response.CodeConflict, "同级已存在同名部门")
	}

	d := model.Dept{
		ParentID: req.ParentID, DeptName: req.DeptName, DeptCode: req.DeptCode,
		LeaderID: req.LeaderID, SortNo: req.SortNo, Remark: req.Remark, Status: 1,
	}
	if err := s.db.Create(&d).Error; err != nil {
		return 0, utils.WrapAppError(response.CodeServerError, "创建部门失败", err)
	}
	return d.ID, nil
}

// DeptQuery 部门平铺列表查询条件：关键字匹配名称或编码，可按状态过滤
type DeptQuery struct {
	utils.PageQuery
	Keyword string `form:"keyword"`
	Status  *int8  `form:"status"`
}

// List 平铺分页查询：树形展示用 Tree，列表页与导出用 List
func (s *DeptService) List(q *DeptQuery) ([]model.Dept, int64, error) {
	q.Normalize()

	tx := s.db.Model(&model.Dept{})
	if q.Keyword != "" {
		kw := "%" + q.Keyword + "%"
		tx = tx.Where("dept_name LIKE ? OR dept_code LIKE ?", kw, kw)
	}
	if q.Status != nil {
		tx = tx.Where("status = ?", *q.Status)
	}

	var total int64
	if err := tx.Count(&total).Error; err != nil {
		return nil, 0, utils.WrapAppError(response.CodeServerError, "查询部门总数失败", err)
	}

	var rows []model.Dept
	if err := tx.Order("sort_no ASC, id ASC").
		Offset(q.Offset()).Limit(q.PageSize).Find(&rows).Error; err != nil {
		return nil, 0, utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}
	return rows, total, nil
}

// Detail 部门详情：返回单条记录与该部门的父级名称
func (s *DeptService) Detail(id uint64) (*model.Dept, string, error) {
	var d model.Dept
	if err := s.db.Where("id = ?", id).First(&d).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, "", utils.NewAppError(response.CodeNotFound, "部门不存在")
		}
		return nil, "", utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}

	parentName := ""
	if d.ParentID != nil {
		var p model.Dept
		if err := s.db.Where("id = ?", *d.ParentID).First(&p).Error; err == nil {
			parentName = p.DeptName
		}
	}
	return &d, parentName, nil
}

// isDescendant 判断 targetID 是否为 rootID 的后代，用于阻止“把父级改到自己下级”形成环
func (s *DeptService) isDescendant(rootID, targetID uint64) (bool, error) {
	cur := targetID
	for i := 0; i < 32; i++ { // 层级上限保护，避免脏数据造成死循环
		var d model.Dept
		if err := s.db.Where("id = ?", cur).First(&d).Error; err != nil {
			if errors.Is(err, gorm.ErrRecordNotFound) {
				return false, nil
			}
			return false, err
		}
		if d.ParentID == nil {
			return false, nil
		}
		if *d.ParentID == rootID {
			return true, nil
		}
		cur = *d.ParentID
	}
	return false, nil
}

// Update 修改部门：编码与同级名称仍需唯一，且不允许把上级调整为自己的下级
func (s *DeptService) Update(id uint64, req *CreateDeptReq) error {
	var d model.Dept
	if err := s.db.Where("id = ?", id).First(&d).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return utils.NewAppError(response.CodeNotFound, "部门不存在")
		}
		return utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}

	if req.ParentID != nil {
		if *req.ParentID == id {
			return utils.NewAppError(response.CodeParamInvalid, "上级部门不能是自身")
		}
		var parentCount int64
		if err := s.db.Model(&model.Dept{}).Where("id = ?", *req.ParentID).
			Count(&parentCount).Error; err != nil {
			return utils.WrapAppError(response.CodeServerError, "查询上级部门失败", err)
		}
		if parentCount == 0 {
			return utils.NewAppError(response.CodeParamInvalid, "上级部门不存在")
		}
		desc, err := s.isDescendant(id, *req.ParentID)
		if err != nil {
			return utils.WrapAppError(response.CodeServerError, "查询部门层级失败", err)
		}
		if desc {
			return utils.NewAppError(response.CodeParamInvalid, "上级部门不能是当前部门的下级")
		}
	}

	var dup int64
	if err := s.db.Model(&model.Dept{}).
		Where("dept_code = ? AND id <> ?", req.DeptCode, id).Count(&dup).Error; err != nil {
		return utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}
	if dup > 0 {
		return utils.NewAppError(response.CodeConflict, "部门编码已存在")
	}

	scope := s.db.Model(&model.Dept{}).Where("dept_name = ? AND id <> ?", req.DeptName, id)
	if req.ParentID != nil {
		scope = scope.Where("parent_id = ?", *req.ParentID)
	} else {
		scope = scope.Where("parent_id IS NULL")
	}
	var sameName int64
	if err := scope.Count(&sameName).Error; err != nil {
		return utils.WrapAppError(response.CodeServerError, "查询部门失败", err)
	}
	if sameName > 0 {
		return utils.NewAppError(response.CodeConflict, "同级已存在同名部门")
	}

	fields := map[string]any{
		"parent_id": req.ParentID, "dept_name": req.DeptName, "dept_code": req.DeptCode,
		"leader_id": req.LeaderID, "sort_no": req.SortNo, "remark": req.Remark,
	}
	if err := s.db.Model(&model.Dept{}).Where("id = ?", id).Updates(fields).Error; err != nil {
		return utils.WrapAppError(response.CodeServerError, "修改部门失败", err)
	}
	return nil
}

// Remove 删除部门：事务内校验并删除；存在子部门或成员时整体拒绝
func (s *DeptService) Remove(id uint64) error {
	// 校验与删除放在同一事务内，避免校验通过后被并发新增子部门
	return s.db.Transaction(func(tx *gorm.DB) error {
		var children int64
		if e := tx.Model(&model.Dept{}).Where("parent_id = ?", id).
			Count(&children).Error; e != nil {
			return utils.WrapAppError(response.CodeServerError, "查询子部门失败", e)
		}
		if children > 0 {
			return utils.NewAppError(response.CodeConflict, "存在子部门，无法删除")
		}

		var members int64
		if e := tx.Model(&model.User{}).Where("dept_id = ?", id).
			Count(&members).Error; e != nil {
			return utils.WrapAppError(response.CodeServerError, "查询部门成员失败", e)
		}
		if members > 0 {
			return utils.NewAppError(response.CodeConflict, "部门下仍有成员，无法删除")
		}

		var d model.Dept
		if e := tx.Where("id = ?", id).First(&d).Error; e != nil {
			if errors.Is(e, gorm.ErrRecordNotFound) {
				return utils.NewAppError(response.CodeNotFound, "部门不存在")
			}
			return utils.WrapAppError(response.CodeServerError, "查询部门失败", e)
		}
		return tx.Delete(&model.Dept{}, id).Error
	})
}
```

### 9.2 系统配置

**存储与取值：**参数以键值对存放，用 value_type 标注类型（1 字符串、2 整数、3 布尔），用 category 分类（base、security、log 等），is_system 标记系统内置项。读取走内存缓存：首次访问时全量加载，之后命中缓存直接返回，避免每次请求都查表；修改成功后立即刷新缓存，做到“改完即生效”，无需重启服务。

**写入校验：**修改前按 value_type 校验取值格式——整数类型的参数传非数字直接返回 400，布尔类型只接受 true/false，把错误拦在入库之前，避免后续读取时类型转换失败。除单条修改外还提供批量修改接口：多条参数放在同一事务内逐条校验与写入，只要有一项不合法就整体回滚，缓存也不会被局部刷新，保证“外部看到的参数集合始终是一致的一份”。

```go
package model

import (
	"time"

	"gorm.io/gorm"
)

// SysConfig 系统参数：键值对存储，value_type 决定读取时的转换方式
type SysConfig struct {
	ID          uint64         `gorm:"primaryKey;autoIncrement" json:"id"`
	ConfigKey   string         `gorm:"size:100;uniqueIndex;not null" json:"configKey"`
	ConfigName  string         `gorm:"size:100;not null" json:"configName"`
	ConfigValue string         `gorm:"size:2000" json:"configValue"`
	ValueType   int8           `gorm:"not null;default:1" json:"valueType"`
	Category    string         `gorm:"size:32;not null;default:base;index" json:"category"`
	IsSystem    int8           `gorm:"not null;default:0" json:"isSystem"`
	Status      int8           `gorm:"not null;default:1;index" json:"status"`
	Remark      string         `gorm:"size:255" json:"remark"`
	CreatedAt   time.Time      `json:"createdAt"`
	UpdatedAt   time.Time      `json:"updatedAt"`
	DeletedAt   gorm.DeletedAt `gorm:"index" json:"-"`
}

func (SysConfig) TableName() string { return "sys_config" }
```

```go
package service

import (
	"errors"
	"strconv"
	"sync"

	"gorm.io/gorm"

	"industrial_admin/model"
	"industrial_admin/response"
	"industrial_admin/utils"
)

// 参数缓存：启动后首次读取时加载，修改后刷新，避免高频查询配置表
var (
	cfgCacheMu sync.RWMutex
	cfgCache   = make(map[string]string)
	cfgLoaded  bool
)

type ConfigService struct{ db *gorm.DB }

func NewConfigService(database *gorm.DB) *ConfigService { return &ConfigService{db: database} }

// Reload 全量刷新缓存
func (s *ConfigService) Reload() error {
	var rows []model.SysConfig
	if err := s.db.Where("status = 1").Find(&rows).Error; err != nil {
		return utils.WrapAppError(response.CodeServerError, "加载参数失败", err)
	}

	cfgCacheMu.Lock()
	cfgCache = make(map[string]string, len(rows))
	for _, r := range rows {
		cfgCache[r.ConfigKey] = r.ConfigValue
	}
	cfgLoaded = true
	cfgCacheMu.Unlock()
	return nil
}

// Get 按键读取参数，未配置时返回默认值
func (s *ConfigService) Get(key, def string) string {
	cfgCacheMu.RLock()
	if cfgLoaded {
		v, ok := cfgCache[key]
		cfgCacheMu.RUnlock()
		if ok {
			return v
		}
		return def
	}
	cfgCacheMu.RUnlock()

	if err := s.Reload(); err != nil {
		return def
	}
	cfgCacheMu.RLock()
	defer cfgCacheMu.RUnlock()
	if v, ok := cfgCache[key]; ok {
		return v
	}
	return def
}

// List 按分类查询参数，分类为空则返回全部
func (s *ConfigService) List(category string) ([]model.SysConfig, error) {
	tx := s.db.Model(&model.SysConfig{})
	if category != "" {
		tx = tx.Where("category = ?", category)
	}
	var rows []model.SysConfig
	if err := tx.Order("category ASC, id ASC").Find(&rows).Error; err != nil {
		return nil, utils.WrapAppError(response.CodeServerError, "查询参数失败", err)
	}
	return rows, nil
}

// Detail 按标识读取单条参数，供编辑弹窗回显
func (s *ConfigService) Detail(id uint64) (*model.SysConfig, error) {
	var row model.SysConfig
	if err := s.db.Where("id = ?", id).First(&row).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, utils.NewAppError(response.CodeNotFound, "参数不存在")
		}
		return nil, utils.WrapAppError(response.CodeServerError, "查询参数失败", err)
	}
	return &row, nil
}

// checkValue 按 value_type 校验取值格式：1 字符串、2 整数、3 布尔
func checkValue(row *model.SysConfig, value string) error {
	switch row.ValueType {
	case 2:
		if _, err := strconv.Atoi(value); err != nil {
			return utils.NewAppError(response.CodeParamInvalid, "参数 "+row.ConfigKey+" 必须为整数")
		}
	case 3:
		if value != "true" && value != "false" {
			return utils.NewAppError(response.CodeParamInvalid, "参数 "+row.ConfigKey+" 必须为 true 或 false")
		}
	}
	return nil
}

// Update 修改参数值：按 value_type 校验格式，成功后刷新缓存
func (s *ConfigService) Update(id uint64, value string) error {
	var row model.SysConfig
	if err := s.db.Where("id = ?", id).First(&row).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return utils.NewAppError(response.CodeNotFound, "参数不存在")
		}
		return utils.WrapAppError(response.CodeServerError, "查询参数失败", err)
	}
	if err := checkValue(&row, value); err != nil {
		return err
	}

	if err := s.db.Model(&model.SysConfig{}).Where("id = ?", id).
		Updates(map[string]any{"config_value": value}).Error; err != nil {
		return utils.WrapAppError(response.CodeServerError, "更新参数失败", err)
	}

	return s.Reload()
}

// ConfigItem 批量修改的单条入参
type ConfigItem struct {
	ID    uint64 `json:"id" binding:"required"`
	Value string `json:"value"`
}

// BatchUpdate 批量修改：一次事务内逐条校验并写入，全部成功才提交并刷新缓存
func (s *ConfigService) BatchUpdate(items []ConfigItem) (int, error) {
	if len(items) == 0 {
		return 0, utils.NewAppError(response.CodeParamInvalid, "待修改参数不能为空")
	}

	updated := 0
	err := s.db.Transaction(func(tx *gorm.DB) error {
		for _, it := range items {
			var row model.SysConfig
			if e := tx.Where("id = ?", it.ID).First(&row).Error; e != nil {
				if errors.Is(e, gorm.ErrRecordNotFound) {
					return utils.NewAppError(response.CodeNotFound, "参数不存在")
				}
				return e
			}
			if e := checkValue(&row, it.Value); e != nil {
				return e
			}
			if e := tx.Model(&model.SysConfig{}).Where("id = ?", it.ID).
				Updates(map[string]any{"config_value": it.Value}).Error; e != nil {
				return e
			}
			updated++
		}
		return nil
	})
	if err != nil {
		var ae *utils.AppError
		if errors.As(err, &ae) {
			return 0, ae
		}
		return 0, utils.WrapAppError(response.CodeServerError, "批量修改参数失败", err)
	}

	if err := s.Reload(); err != nil {
		return 0, err
	}
	return updated, nil
}
```

### 9.3 操作日志

**两类日志来源：**系统里的日志由两处产生。一是操作留痕中间件，挂在全部受保护写接口上，记录请求方法、URL、IP、耗时与响应状态，回答“谁在什么时候调了哪个接口”；二是业务层显式写入，在新增用户、角色授权、改配置、部门增删等动作成功后补一条，带上业务语义明细（如“角色标识 7，授权 2 个权限点”），回答“具体改了什么”。两者互补，前者保证无遗漏，后者保证可读。

**检索设计：**日志表只追加、不更新、不软删除，查询接口支持按操作人、模块、结果、时间范围与关键字五类条件组合过滤，并按主键倒序分页返回，最新记录在第一页。时间参数按 yyyy-MM-dd HH:mm:ss 解析，解析失败即忽略该条件，不因为一个格式错误的时间把整次查询打断。查询本身不写日志，否则会产生日志记录日志的自我循环。

```go
package service

import (
	"time"

	"gorm.io/gorm"

	"industrial_admin/model"
	"industrial_admin/response"
	"industrial_admin/utils"
)

type LogService struct{ db *gorm.DB }

func NewLogService(database *gorm.DB) *LogService { return &LogService{db: database} }

// LogQuery 日志检索条件：操作人、模块、结果、时间范围与关键字
type LogQuery struct {
	utils.PageQuery
	Username  string `form:"username"`
	Module    string `form:"module"`
	Result    *int8  `form:"result"`
	StartTime string `form:"startTime"`
	EndTime   string `form:"endTime"`
	Keyword   string `form:"keyword"`
}

func (s *LogService) List(q *LogQuery) ([]model.OperationLog, int64, error) {
	q.Normalize()

	tx := s.db.Model(&model.OperationLog{})
	if q.Username != "" {
		tx = tx.Where("username = ?", q.Username)
	}
	if q.Module != "" {
		tx = tx.Where("module = ?", q.Module)
	}
	if q.Result != nil {
		tx = tx.Where("result = ?", *q.Result)
	}
	if q.StartTime != "" {
		if t, err := time.Parse("2006-01-02 15:04:05", q.StartTime); err == nil {
			tx = tx.Where("created_at >= ?", t)
		}
	}
	if q.EndTime != "" {
		if t, err := time.Parse("2006-01-02 15:04:05", q.EndTime); err == nil {
			tx = tx.Where("created_at <= ?", t)
		}
	}
	if q.Keyword != "" {
		kw := "%" + q.Keyword + "%"
		tx = tx.Where("url LIKE ? OR error_msg LIKE ?", kw, kw)
	}

	var total int64
	if err := tx.Count(&total).Error; err != nil {
		return nil, 0, utils.WrapAppError(response.CodeServerError, "查询日志总数失败", err)
	}

	var rows []model.OperationLog
	if err := tx.Order("id DESC").Offset(q.Offset()).Limit(q.PageSize).Find(&rows).Error; err != nil {
		return nil, 0, utils.WrapAppError(response.CodeServerError, "查询日志失败", err)
	}
	return rows, total, nil
}
```

### 9.6 本章小结

三个模块的共同点是把“结构约束”写进代码而不是只写进文档：部门树用防环校验保证结构合法，配置用类型校验保证取值合法，日志用只追加与双来源保证记录完整。事务的使用也不局限于“多条 SQL 一起提交”，更在于把“校验 + 写入”这两步圈在一起，避免校验通过后状态被并发改写。这三点是后续测试与验收时最容易被检查到的部分，也是本模块值得在报告中着重说明的实现取舍。
